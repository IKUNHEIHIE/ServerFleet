// W06 临时验收脚本（reports/_review/w06，不提交）：
// 1) 在 127.0.0.1 起静态服务托管隔离预览；2) 用 Chrome DevTools Protocol 发真实 Tab/Shift+Tab/Enter/Space/Escape
//    与真实鼠标事件（Input.dispatchKeyEvent / Input.dispatchMouseEvent），不是 jsdom、不是 element.focus()；
// 3) 驱动 window.__qa 切换来源/settle 请求，记录每步 activeElement、首个提交文本与截图；4) 结束后关闭本脚本启动的 Chrome 与 HTTP 服务。
import { spawn, execSync } from "node:child_process";
import { createServer } from "node:http";
import { mkdirSync, readFileSync, writeFileSync } from "node:fs";
import path from "node:path";

const CHROME = "C:\\Program Files\\Google\\Chrome\\Application\\chrome.exe";
const HTTP_PORT = 8461;
const CDP_PORT = 9461;
const OUT = path.resolve("reports/_review/w06");
const DIST = path.resolve("reports/_preview/dist");
const PAGE = `http://127.0.0.1:${HTTP_PORT}/index.html`;
const sleep = ms => new Promise(resolve => setTimeout(resolve, ms));

mkdirSync(OUT, { recursive: true });

const MIME = { ".html": "text/html; charset=utf-8", ".js": "text/javascript; charset=utf-8", ".css": "text/css; charset=utf-8", ".png": "image/png" };
const server = createServer((request, response) => {
  const url = request.url === "/" ? "/index.html" : request.url.split("?")[0];
  const file = path.join(DIST, path.normalize(url).replace(/^([\\/])+/, ""));
  try {
    const body = readFileSync(file);
    response.writeHead(200, { "content-type": MIME[path.extname(file)] ?? "application/octet-stream" });
    response.end(body);
  } catch {
    response.writeHead(404);
    response.end("not found");
  }
});

const chrome = spawn(

    "--disable-gpu",
    "--hide-scrollbars",
    `--remote-debugging-port=${CDP_PORT}`,
    `--user-data-dir=${path.join(OUT, "profile")}`,
    "--window-size=1280,900",
    PAGE,
  ],
  { stdio: "ignore" },
);

class Cdp {
  constructor(socket) {
    this.socket = socket;
    this.nextId = 1;
    this.pending = new Map();
    socket.addEventListener("message", event => {
      const message = JSON.parse(event.data);
      if (message.id && this.pending.has(message.id)) {
        const { resolve, reject } = this.pending.get(message.id);
        this.pending.delete(message.id);
        message.error ? reject(new Error(JSON.stringify(message.error))) : resolve(message.result);
      }
    });
  }
  send(method, params = {}) {
    const id = this.nextId++;
    this.socket.send(JSON.stringify({ id, method, params }));
    return new Promise((resolve, reject) => this.pending.set(id, { resolve, reject }));
  }
}

async function findTarget() {
  for (let attempt = 0; attempt < 40; attempt += 1) {
    try {
      const response = await fetch(`http://127.0.0.1:${CDP_PORT}/json/list`);
      const targets = await response.json();
      const page = targets.find(target => target.type === "page" && target.url.startsWith("http://127.0.0.1"));
      if (page?.webSocketDebuggerUrl) return page.webSocketDebuggerUrl;
    } catch {
      /* not ready yet */
    }
    await sleep(250);
  }
  throw new Error("DevTools target not available");
}

const results = [];
const record = (name, pass, details) => {
  results.push({ name, pass, details });
  console.log(`${pass ? "PASS" : "FAIL"}  ${name}${details ? ` — ${details}` : ""}`);
};

async function main() {
  const head = execSync("git rev-parse HEAD", { encoding: "utf8" }).trim();
  const wsUrl = await findTarget();
  const socket = new WebSocket(wsUrl);
  await new Promise((resolve, reject) => {
    socket.addEventListener("open", resolve, { once: true });
    socket.addEventListener("error", reject, { once: true });
  });
  const cdp = new Cdp(socket);
  await cdp.send("Page.enable");
  await cdp.send("Runtime.enable");

  const evaluate = async expression => {
    const result = await cdp.send("Runtime.evaluate", { expression, returnByValue: true, awaitPromise: false });
    if (result.exceptionDetails) {
      throw new Error(`eval failed: ${result.exceptionDetails.exception?.description ?? result.exceptionDetails.text} :: ${expression.slice(0, 120)}`);
    }
    return result.result?.value;
  };
  /** 调用 window.__qa 的方法（自动补括号，避免调用约定歧义）。 */
  const qa = (method, ...args) =>
    evaluate(`window.__qa && window.__qa.${method}(${args.map(arg => JSON.stringify(arg)).join(",")})`);
  const commits = () => evaluate("window.__qa ? window.__qa.commits.slice() : []");
  const key = async (k, code, vk, modifiers = 0) => {
    await cdp.send("Input.dispatchKeyEvent", {
      type: "rawKeyDown",
      key: k,
      code,
      windowsVirtualKeyCode: vk,
      nativeVirtualKeyCode: vk,
      modifiers,
    });
    await cdp.send("Input.dispatchKeyEvent", {
      type: "keyUp",
      key: k,
      code,
      windowsVirtualKeyCode: vk,
      nativeVirtualKeyCode: vk,
      modifiers,
    });
    await sleep(160);
  };
  const tab = () => key("Tab", "Tab", 9);
  const shiftTab = () => key("Tab", "Tab", 9, 8);
  /** Enter 需要带 text 的 keyDown 才会触发原生按钮激活（rawKeyDown 只对自定义 onKeyDown 生效）。 */
  const enter = async () => {
    await cdp.send("Input.dispatchKeyEvent", {
      type: "keyDown",
      key: "Enter",
      code: "Enter",
      text: "\r",
      unmodifiedText: "\r",
      windowsVirtualKeyCode: 13,
      nativeVirtualKeyCode: 13,
    });
    await cdp.send("Input.dispatchKeyEvent", {
      type: "keyUp",
      key: "Enter",
      code: "Enter",
      windowsVirtualKeyCode: 13,
      nativeVirtualKeyCode: 13,
    });
    await sleep(200);
  };
  const space = () => key(" ", "Space", 32);
  const escape = () => key("Escape", "Escape", 27);
  const clickAt = async (x, y) => {
    await cdp.send("Input.dispatchMouseEvent", { type: "mousePressed", x, y, button: "left", clickCount: 1 });
    await cdp.send("Input.dispatchMouseEvent", { type: "mouseReleased", x, y, button: "left", clickCount: 1 });
    await sleep(250);
  };
  const clickSelector = async selector => {
    const rect = await evaluate(`(() => { const el = document.querySelector(${JSON.stringify(selector)}); if (!el) return null; const r = el.getBoundingClientRect(); return { x: r.left + r.width / 2, y: r.top + r.height / 2 }; })()`);
    if (!rect) throw new Error(`selector not found: ${selector}`);
    await clickAt(rect.x, rect.y);
  };
  const clickText = async (tag, text) => {
    const rect = await evaluate(`(() => { const el = [...document.querySelectorAll(${JSON.stringify(tag)})].find(x => (x.textContent || "").trim() === ${JSON.stringify(text)}); if (!el) return null; const r = el.getBoundingClientRect(); return { x: r.left + r.width / 2, y: r.top + r.height / 2 }; })()`);
    if (!rect) throw new Error(`button not found: ${text}`);
    await clickAt(rect.x, rect.y);
  };
  const lastRequestIndex = async () => (await qa("requests")).length - 1;
  const commitWait = () => sleep(260);

  for (let attempt = 0; attempt < 40; attempt += 1) {
    if ((await evaluate("document.readyState")) === "complete") break;
    await sleep(150);
  }
  let mounted = false;
  for (let attempt = 0; attempt < 40; attempt += 1) {
    mounted = (await evaluate("typeof window.__qa === 'object'")) === true;
    if (mounted) break;
    await sleep(150);
  }
  if (!mounted) {
    const diag = await evaluate(
      "({ href: location.href, ready: document.readyState, rootChildren: document.getElementById('root')?.children.length ?? -1, text: (document.body.innerText || '').slice(0, 120) })",
    );
    throw new Error(`preview not mounted: ${JSON.stringify(diag)}`);
  }
  await sleep(200);

  const version = await cdp.send("Browser.getVersion");
  await cdp.send("Emulation.setDeviceMetricsOverride", { width: 1280, height: 900, deviceScaleFactor: 1, mobile: false });
  await sleep(250);

  // C1 首次加载
  const initial = await qa("snapshot");
  record(
    "C1 首次加载显示 loading，无表格",
    Boolean(initial?.hasLoading) && !initial?.hasTable && initial.body.includes("正在加载节点"),
    `hasLoading=${initial?.hasLoading} hasTable=${initial?.hasTable}`,
  );
  await cdp.send("Page.captureScreenshot", { format: "png" }).then(shot => {
    writeFileSync(path.join(OUT, "loading-state.png"), Buffer.from(shot.data, "base64"));
  });

  // 成功
  await qa("settle", await lastRequestIndex(), "success");
  await commitWait();
  const ready = await qa("snapshot");
  record(
    "C2 成功后展示节点与计数",
    Boolean(ready?.hasTable) && ready.body.includes("共 8 个节点") && ready.body.includes("aws-tokyo-01"),
    `hasTable=${ready?.hasTable}`,
  );
  record(
    "C3 固定时钟下指标质量（真实 0% 与陈旧可见时间）",
    ready.body.includes("0%") && ready.body.includes("陈旧 · 采样于"),
    `has0=${ready.body.includes("0%")} hasStale=${ready.body.includes("陈旧 · 采样于")}`,
  );

  // C4 真实 Tab 遍历
  await evaluate("document.activeElement && document.activeElement.blur()");
  const tabStops = [];
  for (let step = 1; step <= 8; step += 1) {
    await tab();
    const focus = await qa("focus");
    tabStops.push({ step, ...focus });
    if (focus.tag === "TR") break;
  }
  record(
    "C4 真实 Tab：搜索框 → 状态筛选 → 表格行（禁用的清空按钮被跳过）",
    tabStops.length === 3 && tabStops[0].tag === "INPUT" && tabStops[1].tag === "SELECT" && tabStops[2].tag === "TR",
    tabStops.map(stop => `${stop.step}:<${stop.tag}>`).join(" "),
  );

  // C5 Enter 激活 → 详情聚焦
  await enter();
  let focus = await qa("focus");
  record(
    "C5 Enter 激活行后焦点进入详情区",
    focus.tag === "SECTION" && focus.heading === "节点详情",
    `<${focus.tag}> heading=${focus.heading}`,
  );
  await cdp.send("Page.captureScreenshot", { format: "png" }).then(shot => {
    writeFileSync(path.join(OUT, "focus-detail-1280.png"), Buffer.from(shot.data, "base64"));
  });

  // C6 Tab → 关闭按钮；Tab 离开；Shift+Tab 返回；Escape 回行
  await tab();
  const afterTab = await qa("focus");
  await tab();
  const tabOut = await qa("focus");
  await shiftTab();
  const back = await qa("focus");
  await escape();
  focus = await qa("focus");
  const afterEscape = await qa("snapshot");
  record(
    "C6 Tab/Shift+Tab/Escape：详情内只有关闭按钮，可离开可返回，Escape 回到来源行",
    afterTab.tag === "BUTTON" &&
      afterTab.text === "关闭详情" &&
      tabOut.tag === "BODY" &&
      back.tag === "BUTTON" &&
      focus.tag === "TR" &&
      !afterEscape.hasDetail,
    `tab=<${afterTab.tag}:${afterTab.text}> out=<${tabOut.tag}> back=<${back.tag}> esc=<${focus.tag}>`,
  );

  // C7 Space 激活
  await space();
  focus = await qa("focus");
  record(
    "C7 Space 激活同一行同样立即聚焦详情",
    focus.tag === "SECTION" && focus.heading === "节点详情",
    `<${focus.tag}>`,
  );
  await escape();

  // C8 真实鼠标点击行
  await clickSelector("tbody tr");
  const afterClick = await qa("focus");
  const clickState = await qa("snapshot");
  record(
    "C8 真实鼠标点击行打开详情且不搬动焦点",
    clickState.hasDetail && afterClick.tag === "TR",
    `focus=<${afterClick.tag}>`,
  );
  await escape();

  // C9 失败与重试（鼠标）：换到 B 产生一次新的 pending 请求，再让它失败
  await qa("setSource", "b");
  await commitWait();
  await qa("settle", await lastRequestIndex(), "failure");
  await commitWait();
  const failed = await qa("snapshot");
  record(
    "C9 失败显示固定安全提示（无异常原文）",
    Boolean(failed?.hasAlert) && failed.body.includes("节点数据暂时无法加载，请稍后重试。"),
    `hasAlert=${failed?.hasAlert}`,
  );
  await cdp.send("Page.captureScreenshot", { format: "png" }).then(shot => {
    writeFileSync(path.join(OUT, "error-state.png"), Buffer.from(shot.data, "base64"));
  });
  await clickText("button", "重新加载");
  const retrying = await qa("snapshot");
  record(
    "C10 鼠标点击重新加载立即回到 loading，无节点/计数/详情",
    Boolean(retrying?.hasLoading) && !retrying.hasTable && !retrying.hasDetail,
    `hasLoading=${retrying?.hasLoading} hasTable=${retrying?.hasTable}`,
  );
  await qa("settle", await lastRequestIndex(), "success");
  await commitWait();

  // C11 键盘重试：回到 A 产生新请求并失败，Tab 到按钮，Enter
  await qa("setSource", "a");
  await commitWait();
  await qa("settle", await lastRequestIndex(), "failure");
  await commitWait();
  await evaluate("document.activeElement && document.activeElement.blur()");
  await tab();
  focus = await qa("focus");
  await enter();
  const retry2 = await qa("snapshot");
  record(
    "C11 键盘 Tab+Enter 触发重试同样进入 loading",
    focus.tag === "BUTTON" && focus.text === "重新加载" && Boolean(retry2?.hasLoading),
    `button=<${focus.tag}:${focus.text}> hasLoading=${retry2?.hasLoading}`,
  );
  await qa("settle", await lastRequestIndex(), "success");
  await commitWait();

  // C12 A → B 首提交
  await qa("clearCommits");
  await qa("setSource", "b");
  await commitWait();
  const commitsA = await commits();
  const first = commitsA?.[0] ?? "";
  const switchState = await qa("snapshot");
  record(
    "C12 A→B 首个提交即为 loading，不含旧节点/计数",
    first.includes("正在加载节点") && !first.includes("aws-tokyo-01") && !first.includes("共 8 个节点") && !switchState.hasTable,
    `firstCommitLen=${first.length}`,
  );
  await qa("settle", await lastRequestIndex(), "success");
  await commitWait();
  const bReady = await qa("snapshot");
  record("C13 B 成功后展示 B 的节点", bReady.body.includes("gcp-singapore-01") && !bReady.body.includes("aws-tokyo-01"), "");

  // C14 迟到响应 + 无后台抢焦点
  await qa("setSource", "a");
  await commitWait();
  const lateAIndex = await lastRequestIndex();
  await qa("setSource", "b");
  await commitWait();
  await qa("settle", await lastRequestIndex(), "success");
  await commitWait();
  await clickSelector("input[type='search']");
  const focusBeforeLate = await qa("focus");
  await qa("settle", lateAIndex, "success");
  await commitWait();
  const afterLate = await qa("snapshot");
  const focusAfterLate = await qa("focus");
  const lateRequests = await qa("requests");
  record(
    "C14 旧请求迟到成功不覆盖 B，且不抢走搜索框焦点",
    !afterLate.body.includes("aws-tokyo-01") && focusAfterLate.tag === "INPUT" && lateRequests[lateAIndex].aborted,
    `focusBefore=<${focusBeforeLate.tag}> focusAfter=<${focusAfterLate.tag}> oldAborted=${lateRequests[lateAIndex].aborted}`,
  );

  // C15 A→null→A 首提交（含旧详情）
  await qa("setSource", "a");
  await commitWait();
  await qa("settle", await lastRequestIndex(), "success");
  await commitWait();
  await evaluate("document.activeElement && document.activeElement.blur()");
  for (let i = 0; i < 6; i += 1) {
    await tab();
    if ((await qa("focus")).tag === "TR") break;
  }
  await enter();
  const withDetail = await qa("snapshot");
  await qa("clearCommits");
  await qa("setSource", "none");
  await commitWait();
  const nullState = await qa("snapshot");
  await qa("clearCommits");
  await qa("setSource", "a");
  await commitWait();
  const commitsB = await commits();
  const first2 = commitsB?.[0] ?? "";
  record(
    "C15 A→null→A：无效期隐藏，恢复首个提交即 loading（旧详情不复活）",
    Boolean(withDetail?.hasDetail) &&
      Boolean(nullState?.hasAlert) &&
      !nullState.hasTable &&
      first2.includes("正在加载节点") &&
      !first2.includes("aws-tokyo-01") &&
      !first2.includes("共 8 个节点") &&
      !first2.includes("节点详情"),
    `nullAlert=${nullState?.hasAlert} firstCommitLen=${first2.length}`,
  );
  await qa("settle", await lastRequestIndex(), "success");
  await commitWait();

  // C16 同 key 无效 load 恢复
  await evaluate("document.activeElement && document.activeElement.blur()");
  for (let i = 0; i < 6; i += 1) {
    await tab();
    if ((await qa("focus")).tag === "TR") break;
  }
  await enter();
  await qa("clearCommits");
  await qa("setSource", "broken");
  await commitWait();
  const brokenState = await qa("snapshot");
  await qa("clearCommits");
  await qa("setSource", "a");
  await commitWait();
  const commitsC = await commits();
  const first3 = commitsC?.[0] ?? "";
  record(
    "C16 同 key 无效 load → 恢复有效来源：首个提交即 loading，无旧节点/计数/详情",
    Boolean(brokenState?.hasAlert) &&
      !brokenState.hasTable &&
      first3.includes("正在加载节点") &&
      !first3.includes("aws-tokyo-01") &&
      !first3.includes("共 8 个节点") &&
      !first3.includes("节点详情"),
    `brokenAlert=${brokenState?.hasAlert} firstCommitLen=${first3.length}`,
  );
  await qa("settle", await lastRequestIndex(), "success");
  await commitWait();

  // C17 pending 期间失效：abort + 迟到不覆盖
  await qa("setSource", "b");
  await commitWait();
  const pendingIndex = await lastRequestIndex();
  await qa("setSource", "broken");
  await commitWait();
  const requestsNow = await qa("requests");
  await qa("setSource", "b");
  await commitWait();
  await qa("settle", pendingIndex, "success");
  await commitWait();
  const afterLateOld = await qa("snapshot");
  record(
    "C17 pending 请求在失效时被 abort，迟到成功不覆盖恢复后的 loading",
    requestsNow[pendingIndex].aborted && Boolean(afterLateOld?.hasLoading) && !afterLateOld.hasTable,
    `aborted=${requestsNow[pendingIndex].aborted}`,
  );
  await qa("settle", await lastRequestIndex(), "success");
  await commitWait();

  // C18 详情打开时切换来源的焦点去向
  await evaluate("document.activeElement && document.activeElement.blur()");
  for (let i = 0; i < 6; i += 1) {
    await tab();
    if ((await qa("focus")).tag === "TR") break;
  }
  await enter();
  const detailFocus = await qa("focus");
  await qa("setSource", "a");
  await commitWait();
  const switchFocus = await qa("focus");
  await sleep(400);
  const switchFocusLater = await qa("focus");
  record(
    "C18 详情打开时切换来源：旧详情隐藏，焦点不残留旧详情、不被后台更新抢走",
    detailFocus.tag === "SECTION" && switchFocus.tag !== "SECTION" && switchFocusLater.tag === switchFocus.tag,
    `before=<${detailFocus.tag}> after=<${switchFocus.tag}> later=<${switchFocusLater.tag}>`,
  );
  await qa("settle", await lastRequestIndex(), "success");
  await commitWait();

  // 视觉：键盘焦点可见性（行）
  await evaluate("document.activeElement && document.activeElement.blur()");
  for (let i = 0; i < 6; i += 1) {
    await tab();
    if ((await qa("focus")).tag === "TR") break;
  }
  await cdp.send("Page.captureScreenshot", { format: "png" }).then(shot => {
    writeFileSync(path.join(OUT, "focus-row-1280.png"), Buffer.from(shot.data, "base64"));
  });

  // 视觉：宽屏（详情打开）
  await enter();
  await cdp.send("Page.captureScreenshot", { format: "png" }).then(shot => {
    writeFileSync(path.join(OUT, "wide-1280.png"), Buffer.from(shot.data, "base64"));
  });
  await escape();

  // 视觉：窄屏 420
  await cdp.send("Emulation.setDeviceMetricsOverride", { width: 420, height: 900, deviceScaleFactor: 1, mobile: false });
  await sleep(400);
  const narrowMetrics = await evaluate(
    "(() => { const box = document.querySelector('div.overflow-x-auto'); return { innerWidth: window.innerWidth, htmlScroll: document.documentElement.scrollWidth, bodyScroll: document.body.scrollWidth, boxClient: box ? box.clientWidth : null, boxScroll: box ? box.scrollWidth : null }; })()",
  );
  await cdp.send("Page.captureScreenshot", { format: "png" }).then(shot => {
    writeFileSync(path.join(OUT, "narrow-420.png"), Buffer.from(shot.data, "base64"));
  });
  record(
    "C19 窄屏 420px：页面不横向滚动，表格在卡片内滚动",
    narrowMetrics.htmlScroll <= narrowMetrics.innerWidth &&
      narrowMetrics.boxScroll > narrowMetrics.boxClient,
    `htmlScroll=${narrowMetrics.htmlScroll} innerWidth=${narrowMetrics.innerWidth} box=${narrowMetrics.boxClient}/${narrowMetrics.boxScroll}`,
  );
  await cdp.send("Emulation.setDeviceMetricsOverride", { width: 1280, height: 900, deviceScaleFactor: 1, mobile: false });
  await sleep(300);

  const summary = {
    page: PAGE,
    codeHead: head,
    browser: version,
    keyMechanism:
      "Chrome DevTools Protocol: Input.dispatchKeyEvent (rawKeyDown + keyUp) for Tab/Shift+Tab/Enter/Space/Escape; Input.dispatchMouseEvent for real clicks",
    tabStops,
    results,
    screenshots: [
      "reports/_review/w06/loading-state.png",
      "reports/_review/w06/error-state.png",
      "reports/_review/w06/focus-detail-1280.png",
      "reports/_review/w06/focus-row-1280.png",
      "reports/_review/w06/wide-1280.png",
      "reports/_review/w06/narrow-420.png",
    ],
    narrowMetrics,
  };
  writeFileSync(path.join(OUT, "w06-results.json"), `${JSON.stringify(summary, null, 2)}\n`);
  writeFileSync(path.join(OUT, "w06.log"), `${results.map(r => `${r.pass ? "PASS" : "FAIL"}  ${r.name} — ${r.details ?? ""}`).join("\n")}\n`);

  socket.close();
  const failedCount = results.filter(r => !r.pass).length;
  console.log(`\n${results.length - failedCount}/${results.length} PASS`);
  if (failedCount > 0) process.exitCode = 1;
}

server.listen(HTTP_PORT, "127.0.0.1");
main()
  .catch(error => {
    console.error("FAILED:", error.message);
    process.exitCode = 1;
  })
  .finally(async () => {
    chrome.kill();
