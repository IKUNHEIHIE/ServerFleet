import {spawnSync} from 'node:child_process';
import {mkdirSync,readFileSync,writeFileSync} from 'node:fs';
import {createHash} from 'node:crypto';
import path from 'node:path';
import {root,go,env} from '../scripts/lib.mjs';
const directory=path.join(root,'reports/r1-linux-services');mkdirSync(directory,{recursive:true});
function run(label,exe,args,extra={}){const r=spawnSync(exe,args,{cwd:root,env,encoding:'utf8',timeout:240000,maxBuffer:16*1024*1024,windowsHide:true,...extra});writeFileSync(path.join(directory,label+'.log'),(r.stdout??'')+(r.stderr??'')+(r.error?.message??''));if(r.status!==0)throw Error(label+' failed');return r.stdout.trim();}
const uid=run('uid','wsl.exe',['-d','Ubuntu','--exec','id','-u']);if(uid!=='1000')throw Error('unexpected WSL test identity');
const head=run('head','git',['rev-parse','HEAD']),results=[];
for(const target of ['files','dns','notifications','certificates','publishing']){
 const binary=path.join(directory,target+'.test');
 run(target+'-build',go,['test','-mod=readonly','-c','-o',binary,'./internal/'+target],{env:{...env,GOOS:'linux',GOARCH:'amd64',CGO_ENABLED:'0'}});
 const linux='/mnt/'+binary[0].toLowerCase()+'/'+binary.slice(3).replaceAll('\\','/');
 const output=run(target+'-native','wsl.exe',['-d','Ubuntu','--exec','env','-u','TMPDIR','-u','TMP','-u','TEMP',linux,'-test.v','-test.timeout=180s']);
 const result={target,head,uid,passed:(output.match(/^--- PASS:/gm)??[]).length,skipped:(output.match(/^--- SKIP:/gm)??[]).length,status:'PASS',binarySHA256:createHash('sha256').update(readFileSync(binary)).digest('hex')};
 results.push(result);console.log(JSON.stringify(result));
}
writeFileSync(path.join(directory,'results.json'),JSON.stringify({head,scope:'WSL Linux amd64 native synthetic module tests; no real provider accounts or physical two-machine transfer',results},null,2)+'\n');
