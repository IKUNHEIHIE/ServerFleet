class Cdp {
  constructor(socket) {
    this.socket = socket;
    this.nextId = 1;
    this.pending = new Map();
    socket.addEventListener("message", event => {
      const message = JSON.parse(event.data);
      if (message.id && this.pending.has(message.id)) {
        const { resolve, reject } = this.pending.get(message.id);
        this.pending.delete(message.id);
        message.error ? reject(new Error(JSON.stringify(message.error))) : resolve(message.result);
      }
    });
  }
  send(method, params = {}) {
    const id = this.nextId++;
    this.socket.send(JSON.stringify({ id, method, params }));
    return new Promise((resolve, reject) => this.pending.set(id, { resolve, reject }));
  }
}

async function findTarget() {
  for (let attempt = 0; attempt < 40; attempt += 1) {
    try {
      const response = await fetch(`http://127.0.0.1:${CDP_PORT}/json/list`);
      const targets = await response.json();
      const page = targets.find(target => target.type === "page" && target.url.startsWith("http://127.0.0.1"));
      if (page?.webSocketDebuggerUrl) return page.webSocketDebuggerUrl;
    } catch {
      /* not ready yet */
    }
    await sleep(250);
  }
  throw new Error("DevTools target not available");
}

  await capture("r02-late-failure-1280.png");

  // C24 同 key 失效恢复后旧请求迟到失败
  await qa("setSource", "b");
  await commitWait();
  const staleSameKey = await lastRequestIndex();
  await qa("setSource", "broken");
  await commitWait();
  await qa("setSource", "a");
  await commitWait();
  const freshAfterBroken = await lastRequestIndex();
  await qa("settle", staleSameKey, "failure");
  await commitWait();
  const afterStaleFail = await state();
  traces.lateFailureSameKey = { afterStaleFail };
  record(
    "C24 R02 同 key 失效恢复后旧请求迟到失败：仍保持 loading，不变成错误态",
    Boolean(afterStaleFail?.hasLoading) && !afterStaleFail.hasAlert && !afterStaleFail.hasTable,
    { hasLoading: afterStaleFail?.hasLoading, hasAlert: afterStaleFail?.hasAlert },
  );
  await qa("settle", freshAfterBroken, "success");
  await commitWait();

  // ===== W06-R02：pending 卸载（两个独立请求：迟到成功 / 迟到失败）=====
  const errorsBeforeUnmount = (await evaluate("window.__qa.errors.slice()")).length;
  await qa("setSource", "b");
  await commitWait();
  const unmountSuccessIndex = await lastRequestIndex();
  await qa("setMounted", false);
  await commitWait();
  const afterUnmount = await state();
  const unmountRequests = await qa("requests");
  await qa("settle", unmountSuccessIndex, "success");
  await commitWait();
  const afterUnmountLateSuccess = await state();
  const errorsAfterLateSuccess = (await evaluate("window.__qa.errors.slice()")).length;

  await qa("setMounted", true);
  await commitWait();
  await qa("setSource", "a");
  await commitWait();
  const unmountFailureIndex = await lastRequestIndex();
  await qa("setMounted", false);
  await commitWait();
  await qa("settle", unmountFailureIndex, "failure");
  await commitWait();
  const afterUnmountLateFailure = await state();
  const errorsFinal = (await evaluate("window.__qa.errors.slice()")).length;
  traces.unmount = { afterUnmount, afterUnmountLateSuccess, afterUnmountLateFailure, errorsBeforeUnmount, errorsAfterLateSuccess, errorsFinal };
  record(
    "C25 R02 pending 卸载：请求被 abort，迟到成功/失败均无渲染变化、无未处理拒绝、无抢焦点",
    unmountRequests[unmountSuccessIndex].aborted &&
      !afterUnmount.hasTable &&
      !afterUnmount.hasAlert &&
      !afterUnmount.hasDetail &&
      !afterUnmountLateSuccess.hasTable &&
      !afterUnmountLateFailure.hasTable &&
      !afterUnmountLateFailure.hasAlert &&
      errorsFinal === errorsBeforeUnmount,
    {
      aborted: unmountRequests[unmountSuccessIndex].aborted,
      errorsBeforeUnmount,
      errorsAfterLateSuccess,
      errorsFinal,
    },
  );
  await capture("r02-unmounted.png");

  // 恢复渲染，供截图/复核
  await qa("setMounted", true);
  await commitWait();
  await qa("setSource", "a");
  await commitWait();
  const finalPending = (await qa("requests")).at(-1);
  if (finalPending && finalPending.settled === null) {
    await qa("settle", await lastRequestIndex(), "success");
    await commitWait();
  }

  const summary = {
    page: PAGE,
    codeHead: head,
    browser: version,
    keyMechanism:
      "Chrome DevTools Protocol. Tab/Shift+Tab/Escape/Space: Input.dispatchKeyEvent with type=rawKeyDown + keyUp. Enter: type=keyDown with text=\"\\r\" (rawKeyDown does not activate native buttons) + keyUp. Clicks: Input.dispatchMouseEvent mousePressed/mouseReleased.",
    tabStops,
    results,
    traces,
    screenshots: [
      "reports/_review/w06r1/loading-state.png",
      "reports/_review/w06r1/error-state.png",
      "reports/_review/w06r1/focus-detail-1280.png",
      "reports/_review/w06r1/focus-row-1280.png",
      "reports/_review/w06r1/wide-1280.png",
      "reports/_review/w06r1/narrow-420.png",
      "reports/_review/w06r1/r01-repeat-enter-1280.png",
      "reports/_review/w06r1/r01-close-return-1280.png",
      "reports/_review/w06r1/r02-late-failure-1280.png",
      "reports/_review/w06r1/r02-unmounted.png",
    ],
    narrowMetrics,
  };
  writeFileSync(path.join(OUT, "w06r1-results.json"), `${JSON.stringify(summary, null, 2)}\n`);
  writeFileSync(
    path.join(OUT, "w06r1.log"),
    `${results.map(r => `${r.pass ? "PASS" : "FAIL"}  ${r.name} — ${typeof r.details === "string" ? r.details : JSON.stringify(r.details ?? {})}`).join("\n")}\n`,
  );

  socket.close();
  const failedCount = results.filter(r => !r.pass).length;
  console.log(`\n${results.length - failedCount}/${results.length} PASS`);
  if (failedCount > 0) process.exitCode = 1;
}

server.listen(HTTP_PORT, "127.0.0.1");
main()
  .catch(error => {

    process.exitCode = 1;
  })

    chrome.kill();
    await sleep(300);
    server.close();
  });
