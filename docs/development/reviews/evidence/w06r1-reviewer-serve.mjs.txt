import { createServer } from 'node:http';
import { readFileSync } from 'node:fs';
import path from 'node:path';
const root = path.resolve('reports/_review/w06close/dist');
const server = createServer((req, res) => {
  const pathname = new URL(req.url, 'http://127.0.0.1:8463').pathname;
  const file = path.resolve(root, '.' + (pathname === '/' ? '/index.html' : pathname));
  if (!file.startsWith(root + path.sep)) { res.writeHead(403).end(); return; }
  try {
    const body = readFileSync(file);
    res.writeHead(200, { 'Content-Type': file.endsWith('.js') ? 'text/javascript' : file.endsWith('.css') ? 'text/css' : 'text/html', 'Cache-Control': 'no-store' });
    res.end(body);
  } catch { res.writeHead(404).end(); }
});
server.listen(8463, '127.0.0.1', () => console.log('REVIEW_PREVIEW=http://127.0.0.1:8463'));
process.on('SIGINT', () => server.close(() => process.exit(0)));
