func TestAIScopeRealSQLiteCurrentIdentityRestartAndRevoke(t *testing.T) {
	ctx, cancel := context.WithTimeout(context.Background(), 15*time.Second)
	defer cancel()
	path := filepath.Join(t.TempDir(), "ai.db")
	db := newDB(t, path, true)
	defer db.Close()
	for _, sql := range ai.SessionSchemaStatements() {
		if _, err := db.Exec(ctx, sql); err != nil {
			t.Fatal(err)
		}
	}
	for _, migration := range audit.Migrations() {
		for _, sql := range migration.Statements {
			if _, err := db.Exec(ctx, sql); err != nil {
				t.Fatal(err)
			}
		}
	}
	identity, _ := auth.NewIdentityServiceWithResources(identityStore(t, db), func(_ context.Context, kind, id string) ([]string, error) { return []string{id}, nil }, nil)
	setup, err := identity.PrepareSetup(ctx)
	if err != nil {
		t.Fatal(err)
	}
	if err = identity.Bootstrap(ctx, setup, "admin", "synthetic-long-password"); err != nil {
		t.Fatal(err)
	}
	login, err := identity.Login(ctx, "admin", "synthetic-long-password", "127.0.0.1")
	if err != nil {
		t.Fatal(err)
	}
	_, err = identity.CreateResourceGrant(ctx, login.Secret, login.CSRF, auth.GrantRequest{PrincipalID: login.Identity.SubjectID, ResourceType: "node", ResourceID: "node-a", Permissions: []string{"nodes.read"}})
	if err != nil {
		t.Fatal(err)
	}
	observed, err := identity.Authenticate(ctx, login.Secret, login.CSRF, true)
	if err != nil {
		t.Fatal(err)
	}
	validator, err := audit.NewValidator([]string{"ai.sessions.create", "ai.sessions.grant", "ai.sessions.revoke", "ai.sessions.policy", "ai.sessions.close", "ai.messages.create", "ai.messages.generate", "ai.messages.completed", "ai.messages.unknown"}, audit.DefaultLimits())
	if err != nil {
		t.Fatal(err)
	}
	fail := false
	failMessageCommit, failCompletedCommit := false, false
	service := func(db *ds.DB) *ai.SessionService {
		recorder, err := audit.NewSQLRecorder(db, validator, time.Now)
		if err != nil {
			t.Fatal(err)
		}
		read := func(q ds.Queryer) func(context.Context, string, ...any) ([]byte, error) {
			return func(ctx context.Context, sql string, args ...any) ([]byte, error) {
				var b []byte
				err := q.QueryRow(ctx, sql, args...).Scan(&b)
				return b, err
			}
		}
		store, err := ai.NewSQLSessionStore(ai.SessionSQLBackend{Read: read(db), InTx: func(ctx context.Context, fn func(ai.SessionSQLStatements) error) error {
			return db.InTx(ctx, func(ctx context.Context, tx ds.Tx) error {
				messageMutation, completedMutation := false, false
				ops := ai.SessionSQLStatements{Read: read(tx), Exec: func(ctx context.Context, sql string, args ...any) (int64, error) {
					r, err := tx.Exec(ctx, sql, args...)
					if err != nil {
						return 0, err
					}
					return r.RowsAffected()
				}, AppendAudit: func(ctx context.Context, events []ai.SessionAuditEvent) error {
					for _, e := range events {
						messageMutation = messageMutation || e.Action == "ai.messages.create"
						completedMutation = completedMutation || e.Action == "ai.messages.completed"
						requestID := e.SessionID
						if e.MessageID != "" {
							requestID = e.MessageID
						}
						result, reason, source := e.Result, "", audit.SourceWeb
						if result == "unknown" {
							result, reason, source = "failure", "ai_generation_unknown", audit.SourceSystem
						}
						if err := recorder.AppendTx(ctx, tx, audit.Event{SubjectID: e.SubjectID, Action: e.Action, Result: result, ReasonCode: reason, RequestID: requestID, Source: source, At: e.At, NodeIDs: e.NodeIDs}); err != nil {
							return err
						}
					}
					return nil
				}}
				if err := fn(ops); err != nil {
					return err
				}
				if fail || failMessageCommit && messageMutation || failCompletedCommit && completedMutation {
					return errors.New("injected before commit")
				}
				return nil
			})
		}})
		if err != nil {
			t.Fatal(err)
		}
		access := func(ctx context.Context, _, _ string) (ai.SessionActor, error) {
			if identity.CheckObservationReadOnly(ctx, login.Secret, observed) != nil {
				return ai.SessionActor{}, ai.ErrToolDenied
			}
			return ai.SessionActor{SubjectID: observed.SubjectID, HumanWeb: true, CanFullAccess: true}, nil
		}
		authority := func(ctx context.Context, subject, action string) (ai.CurrentAuthority, error) {
			if identity.Authorize(ctx, auth.Check{SubjectID: subject, Action: action, NodeIDs: []string{"node-a"}}) != nil {
				return ai.CurrentAuthority{SubjectID: subject}, nil
			}
			return ai.CurrentAuthority{SubjectID: subject, AllowedTools: []string{"nodes.read"}, AllowedNodes: []string{"node-a"}}, nil
		}
		result, err := ai.NewSessionService(store, access, authority, []ai.RegisteredTool{{Definition: ai.ToolDefinition{ID: "nodes.read", ReadOnly: true, AutoAllowed: true}, Canonicalize: func(raw json.RawMessage) (json.RawMessage, error) {
			if string(raw) != "{}" {
				return nil, ai.ErrToolDenied
			}
			return raw, nil
		}}})
		if err != nil {
			t.Fatal(err)
		}
		return result
	}
	aiService := service(db)
	session, err := aiService.Create(ctx)
	if err != nil || len(session.AllowedNodeIDs) != 0 {
		t.Fatal(session, err)
	}
	fail = true
	if _, err = aiService.AddGrant(ctx, session.ID, 1, ai.ResourceGrantInput{ResourceType: "node", ResourceID: "node-a", AllowedActions: []string{"nodes.read"}}); err != ai.ErrSessionUnavailable {
		t.Fatal(err)
	}
	fail = false
	session, err = aiService.Get(ctx, session.ID)
	if err != nil || session.ScopeRevision != 1 || len(session.ResourceGrants) != 0 {
		t.Fatal("failed grant changed scope", session, err)
	}
	session, err = aiService.AddGrant(ctx, session.ID, 1, ai.ResourceGrantInput{ResourceType: "node", ResourceID: "node-a", AllowedActions: []string{"nodes.read"}})
	if err != nil {
		t.Fatal(err)
	}
	session, err = aiService.SetPolicy(ctx, session.ID, session.ScopeRevision, ai.LowRisk, 0)
	if err != nil {
		t.Fatal(err)
	}
	_ = db.Close()
	db = newDB(t, path, false)
	defer db.Close()
	identity, _ = auth.NewIdentityServiceWithResources(identityStore(t, db), func(_ context.Context, kind, id string) ([]string, error) { return []string{id}, nil }, nil)
	aiService = service(db)
	call := ai.ToolCall{ToolID: "nodes.read", NodeIDs: []string{"node-a"}, Parameters: []byte(`{}`), ScopeRevision: session.ScopeRevision}
	if assessed, err := aiService.AssessTool(ctx, session.ID, call); err != nil || assessed.RequiresApproval {
		t.Fatal("persisted scope not enforced", assessed, err)
	}
	providerCalls := 0
	conversation := func() *ai.ConversationService {
		result, err := ai.NewConversationService(aiService, ai.ConversationConfig{
			Profile: func(context.Context, string) (ai.ModelProfile, error) {
				return ai.ModelProfile{ID: "fixture", Model: "test-model", CredentialRef: "fixture-ref", Revision: 1}, nil
			},
			Filter: func(_ context.Context, _ ai.SessionView, _ string, text string) (string, error) {
				return strings.ReplaceAll(text, "synthetic-secret", "[redacted]"), nil
			},
			Generate: func(_ context.Context, _ ai.GenerationRequest, emit func(string) error) error {
				providerCalls++
				return emit("fixture text only")
			},
		})
		if err != nil {
			t.Fatal(err)
		}
		return result
	}
	chat := conversation()
	input := ai.MessageInput{Content: "question synthetic-secret", ClientMessageID: "client-one", ScopeRevision: session.ScopeRevision}
	failMessageCommit = true
	if _, err = chat.Submit(ctx, session.ID, "request-one", input); err != ai.ErrSessionUnavailable {
		t.Fatal(err)
	}
	failMessageCommit = false
	if events, err := chat.Events(ctx, session.ID, 0, 100); err != nil || len(events) != 0 {
		t.Fatal("rolled back message emitted an event", events, err)
	}
	message, err := chat.Submit(ctx, session.ID, "request-one", input)
	if err != nil {
		t.Fatal(err)
	}
	failCompletedCommit = true
	if _, err = chat.Generate(ctx, session.ID, message.MessageID); err != ai.ErrGenerationUnknown {
		t.Fatal(err)
	}
	failCompletedCommit = false
	var successfulAnswers int
	if err = db.QueryRow(ctx, "SELECT COUNT(*) FROM audit_events WHERE action='ai.messages.completed'").Scan(&successfulAnswers); err != nil || successfulAnswers != 0 {
		t.Fatal("failed answer audit committed", successfulAnswers, err)
	}
	_ = db.Close()
	db = newDB(t, path, false)
	defer db.Close()
	identity, _ = auth.NewIdentityServiceWithResources(identityStore(t, db), func(_ context.Context, kind, id string) ([]string, error) { return []string{id}, nil }, nil)
	aiService = service(db)
	chat = conversation()
	if _, err = chat.Generate(ctx, session.ID, message.MessageID); err != ai.ErrGenerationUnknown || providerCalls != 1 {
		t.Fatal("restart retried uncertain model call", providerCalls, err)
	}
	if messages, err := chat.Messages(ctx, session.ID, 0, 10); err != nil || len(messages) != 1 || messages[0].RedactedContent != "question [redacted]" {
		t.Fatal("restart lost redacted message or invented answer", messages, err)
	}
	events, err := chat.Events(ctx, session.ID, 0, 100)
	if err != nil || len(events) != 3 || events[2].State != "unknown" {
		t.Fatal("unknown generation not durably recorded", events, err)
	}
	// A REAL HTTPS SSE stream, real Cookie authentication and current final
	// CommitGate. Provider/filter remain explicit doubles, not a model test.
	server := httptest.NewUnstartedServer(nil)
	origin := "https://" + server.Listener.Addr().String()
	authHTTP, err := auth.NewIdentityHTTP(identity, auth.IdentityHTTPConfig{Origin: origin})
	if err != nil {
		t.Fatal(err)
	}
	type observedKey struct{}
	authenticateMessage := func(r *http.Request) (context.Context, error) {
		current, err := authHTTP.AuthenticateRequest(r)
		if err != nil {
			return nil, err
		}
		return context.WithValue(r.Context(), observedKey{}, current), nil
	}
	commitMessage := func(ctx context.Context) error {
		current, ok := ctx.Value(observedKey{}).(auth.Identity)
		if !ok {
			return auth.ErrCredential
		}
		return identity.CommitGate(ctx, login.Secret, current)
	}
	sse, err := ai.NewMessageEventsHTTP(chat, authenticateMessage, commitMessage)
	if err != nil {
		t.Fatal(err)
	}
	messagesHTTP, err := ai.NewMessagesHTTP(chat, authenticateMessage, commitMessage)
	if err != nil { t.Fatal(err) }
	mux := http.NewServeMux()
	mux.Handle("/api/v1/ai/sessions/{sessionId}/events", sse)
	mux.Handle("/api/v1/ai/sessions/{sessionId}/messages", messagesHTTP)
	server.Config.Handler = mux
	server.StartTLS()
	defer server.Close()
	requestMessages := func(method, suffix string, input *ai.MessageInput, csrf bool) (int, []byte) {
		var body []byte
		if input != nil { body, _ = json.Marshal(input) }
		request, err := http.NewRequestWithContext(ctx,method,origin+"/api/v1/ai/sessions/"+session.ID+"/messages"+suffix,strings.NewReader(string(body)))
		if err != nil { t.Fatal(err) }
		request.AddCookie(&http.Cookie{Name:auth.SessionCookieName,Value:login.Secret})
		request.Header.Set("Origin",origin)
		if method == http.MethodPost {
			request.Header.Set("Content-Type","application/json")
			request.Header.Set("Idempotency-Key","request-two")
			if csrf { request.Header.Set("X-CSRF-Token",login.CSRF) }
		}
		response,err := server.Client().Do(request)
		if err != nil { t.Fatal(err) }
		defer response.Body.Close()
		data,err := io.ReadAll(response.Body)
		if err != nil { t.Fatal(err) }
		if strings.Contains(string(data),"synthetic-secret") { t.Fatal("HTTP disclosed raw test secret") }
		return response.StatusCode,data
	}
	open := func(cursor string) *http.Response {
		request, err := http.NewRequestWithContext(ctx, http.MethodGet, origin+"/api/v1/ai/sessions/"+session.ID+"/events", nil)
		if err != nil {
			t.Fatal(err)
		}
		request.AddCookie(&http.Cookie{Name: auth.SessionCookieName, Value: login.Secret})
		request.Header.Set("Origin", origin)
		if cursor != "" {
			request.Header.Set("Last-Event-ID", cursor)
		}
		response, err := server.Client().Do(request)
		if err != nil {
			t.Fatal(err)
		}
		if response.StatusCode != 200 || response.Header.Get("Content-Type") != "text/event-stream" {
			_ = response.Body.Close()
			t.Fatal("SSE not admitted", response.StatusCode)
		}
		return response
	}
	readEvent := func(reader *bufio.Reader) ai.MessageEvent {
		for {
			line, err := reader.ReadString('\n')
			if err != nil {
				t.Fatal(err)
			}
			if strings.HasPrefix(line, "data: ") {
				var event ai.MessageEvent
				if err = json.Unmarshal([]byte(strings.TrimPrefix(line, "data: ")), &event); err != nil {
					t.Fatal(err)
				}
				return event
			}
		}
	}
	first := open("")
	reader := bufio.NewReader(first.Body)
	for i := 1; i <= 3; i++ {
		if e := readEvent(reader); e.Sequence != uint64(i) {
			t.Fatal(e)
		}
	}
	_ = first.Body.Close()
	last := uint64(len(events))
	resumed := open(session.ID + "." + strconv.FormatUint(last, 10))
	defer resumed.Body.Close()
	input.ClientMessageID = "client-two"
	if code,_ := requestMessages(http.MethodPost,"",&input,false);code != 403 { t.Fatal("missing real CSRF accepted",code) }
	failMessageCommit = true
	if code,_ := requestMessages(http.MethodPost,"",&input,true);code != 503 { t.Fatal("failed SQL commit became HTTP success",code) }
	failMessageCommit = false
	code,body := requestMessages(http.MethodPost,"",&input,true)
	var secondMessage ai.Message
	if code != 200 || json.Unmarshal(body,&secondMessage) != nil || secondMessage.Role != "user" || secondMessage.RedactedContent != "question [redacted]" || providerCalls != 1 { t.Fatal("real message submission failed or invoked provider",code) }
	code,replayedBody := requestMessages(http.MethodPost,"",&input,true)
	if code != 200 || string(replayedBody) != string(body) { t.Fatal("real HTTP replay changed persisted message",code) }
	var page struct { Items []ai.Message `json:"items"`;Page struct { NextCursor string `json:"nextCursor"`;HasMore bool `json:"hasMore"` } `json:"page"` }
	code,body = requestMessages(http.MethodGet,"?limit=1",nil,false)
	if code != 200 || json.Unmarshal(body,&page) != nil || len(page.Items) != 1 || page.Items[0].MessageID != message.MessageID || !page.Page.HasMore { t.Fatal("real first page failed",code) }
	cursor := page.Page.NextCursor
	code,body = requestMessages(http.MethodGet,"?limit=1&cursor="+url.QueryEscape(cursor),nil,false)
	if code != 200 || json.Unmarshal(body,&page) != nil || len(page.Items) != 1 || page.Items[0].MessageID != secondMessage.MessageID || page.Page.HasMore { t.Fatal("real cursor page failed",code) }
	reader = bufio.NewReader(resumed.Body)
	if e := readEvent(reader); e.Sequence != last+1 {
		t.Fatal("SSE replayed old events", e)
	}
	if err = identity.RevokeSession(ctx, login.Secret, login.CSRF, observed.SessionID); err != nil {
		t.Fatal(err)
	}
	if code,_ := requestMessages(http.MethodGet,"",nil,false);code != 401 { t.Fatal("revoked real Cookie read messages",code) }
	if tail, err := io.ReadAll(reader); err != nil || strings.Contains(string(tail), "redactedContent") {
		t.Fatal("revoked SSE did not close or leaked content", err)
	}
	if _, err = aiService.AssessTool(ctx, session.ID, call); err != ai.ErrToolDenied {
		t.Fatal("revoked owner session retained tool access", err)
	}
}
