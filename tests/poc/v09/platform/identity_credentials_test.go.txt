 "bytes"
 "context"
 "encoding/json"
 "errors"

 "strings"
 "testing"

 "serverfleet.local/serverfleet/internal/auth"

// Real SQLite encryption/restart/replay evidence. The task gate is a test
// fixture here, NOT evidence of integrated task/approval authorization.
func TestCredentialsRealSQLiteRestartAndUnknownFence(t *testing.T){
 ctx:=context.Background();path:=filepath.Join(t.TempDir(),"credentials.db")
 db:=newDB(t,path,true);defer db.Close()
 identity,_:=auth.NewIdentityService(identityStore(t,db))
 setup,err:=identity.PrepareSetup(ctx);if err!=nil{t.Fatal(err)}
 if err=identity.Bootstrap(ctx,setup,"admin","synthetic-long-password");err!=nil{t.Fatal(err)}
 session,err:=identity.Login(ctx,"admin","synthetic-long-password","127.0.0.1");if err!=nil{t.Fatal(err)}
 keys:=map[string][]byte{"test-key":bytes.Repeat([]byte{71},32)}
 gate:=func(context.Context,auth.CredentialOperation)error{return nil}
 vault,err:=auth.NewCredentialService(identity,"test-key",keys,gate,nil);if err!=nil{t.Fatal(err)}
 const plain="sqlite-synthetic-byok-secret"
 meta,err:=vault.CreateIdempotent(ctx,session.Secret,session.CSRF,"sqlite-credential-create-01",auth.CredentialInput{Name:"Synthetic",Kind:"ai_api_key",Secret:plain,UsageScope:[]string{"ai"}});if err!=nil{t.Fatal(err)}
 op:=auth.CredentialOperation{SubjectID:session.Identity.SubjectID,CredentialID:meta.ID,Module:"ai",Source:"web",OperationID:"provider-test-01",PlanDigest:strings.Repeat("a",64),ExpectedRevision:1,Action:"use"}
 calls:=0
 if err=vault.UseForService(ctx,op,func(_ context.Context,b []byte)error{calls++;if string(b)!=plain{t.Error("wrong credential")};return errors.New("unknown provider result")});err!=auth.ErrCredentialUseUnknown{t.Fatal(err)}
 persisted,_,err:=statements(db).Read(ctx,"SELECT payload,ever_initialized FROM cx_auth_identity WHERE singleton=1")
 if err!=nil||bytes.Contains(persisted,[]byte(plain)){t.Fatal("plaintext at rest",err)}
 var state struct{CredentialKeyUses map[string]uint64};if json.Unmarshal(persisted,&state)!=nil||state.CredentialKeyUses["test-key"]!=1{t.Fatal("durable encryption budget",string(persisted))}
 _=db.Close()
 reopened:=newDB(t,path,false);defer reopened.Close()
 identity,_=auth.NewIdentityService(identityStore(t,reopened));vault,_=auth.NewCredentialService(identity,"test-key",keys,gate,nil)
 listed,err:=vault.List(ctx,session.Secret);if err!=nil||len(listed)!=1||listed[0].Status!="unverified"{t.Fatal(listed,err)}
 if err=vault.UseForService(ctx,op,func(context.Context,[]byte)error{calls++;return nil});err!=auth.ErrCredentialReplay||calls!=1{t.Fatal("restart retried unknown operation",calls,err)}
 op.OperationID="provider-test-02";op.Action="revoke"
 if err=vault.Revoke(ctx,op);err!=nil{t.Fatal(err)}
 op.OperationID="provider-test-03";op.Action="use"
 if err=vault.UseForService(ctx,op,func(context.Context,[]byte)error{t.Fatal("revoked secret used");return nil});err!=auth.ErrDenied{t.Fatal(err)}
}
