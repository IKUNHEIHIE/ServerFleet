package platformpoc

import (
	"context"
	"testing"
	"time"

	"github.com/tetratelabs/wazero"
)

// Small reviewed byte fixtures, not an installed plugin or an ABI proposal.
var wasmHeader = []byte{0, 97, 115, 109, 1, 0, 0, 0}

func wasmFixture(parts ...[]byte) []byte {
	out := append([]byte(nil), wasmHeader...)
	for _, part := range parts {
		out = append(out, part...)
	}
	return out
}

var typeVoid = []byte{1, 4, 1, 96, 0, 0}
var oneFunction = []byte{3, 2, 1, 0}
var exportRun = []byte{7, 7, 1, 3, 114, 117, 110, 0, 0}

func TestWASMMemoryAndUnregisteredHostDenied(t *testing.T) {
	ctx := context.Background()
	runtime := wazero.NewRuntimeWithConfig(ctx, wazero.NewRuntimeConfigInterpreter().WithMemoryLimitPages(2).WithCloseOnContextDone(true))
	defer runtime.Close(ctx)
	// Requests 3*64KiB when the configured per-memory maximum is 2*64KiB.
	oversized := wasmFixture([]byte{5, 3, 1, 0, 3})
	if _, err := runtime.Instantiate(ctx, oversized); err == nil {
		t.Fatal("oversized memory admitted")
	}
	// Imports env.escape; there is no host module and no WASI/fs/network setup.
	hostImport := wasmFixture(typeVoid, []byte{2, 14, 1, 3, 101, 110, 118, 6, 101, 115, 99, 97, 112, 101, 0, 0})
	if _, err := runtime.Instantiate(ctx, hostImport); err == nil {
		t.Fatal("unregistered host capability admitted")
	}
}

func TestWASMInfiniteLoopTerminatesAndRuntimeSurvives(t *testing.T) {
	ctx := context.Background()
	runtime := wazero.NewRuntimeWithConfig(ctx, wazero.NewRuntimeConfigInterpreter().WithMemoryLimitPages(2).WithCloseOnContextDone(true))
	defer runtime.Close(ctx)
	loop := wasmFixture(typeVoid, oneFunction, exportRun, []byte{10, 9, 1, 7, 0, 3, 64, 12, 0, 11, 11})
	module, err := runtime.Instantiate(ctx, loop)
	if err != nil {
		t.Fatal(err)
	}
	deadline, cancel := context.WithTimeout(ctx, 50*time.Millisecond)
	defer cancel()
	started := time.Now()
	_, err = module.ExportedFunction("run").Call(deadline)
	if err == nil || !module.IsClosed() || time.Since(started) > 2*time.Second {
		t.Fatal("infinite loop failed to terminate", err)
	}
	value := wasmFixture([]byte{1, 5, 1, 96, 0, 1, 127}, oneFunction, exportRun, []byte{10, 6, 1, 4, 0, 65, 42, 11})
	healthy, err := runtime.Instantiate(ctx, value)
	if err != nil {
		t.Fatal(err)
	}
	result, err := healthy.ExportedFunction("run").Call(ctx)
	if err != nil || len(result) != 1 || result[0] != 42 {
		t.Fatal(result, err)
	}
}
