type materialSQLReaderKey struct{}
type materialSQLActorKey struct{}
type materialSQLActor struct {
	secret      string
	observation auth.Identity
}

func materialSQLDigest(v any) string {
	raw, _ := json.Marshal(v)
	hash := sha256.Sum256(raw)
	return hex.EncodeToString(hash[:])
}

func TestXrayActualCentralSQLiteMaterialExportFenceAuditAndScope(t *testing.T) {
	ctx, cancel := context.WithTimeout(context.Background(), 45*time.Second)
	defer cancel()
	dbPath := filepath.Join(t.TempDir(), "central-material.db")
	db := newDB(t, dbPath, true)
	defer func() { _ = db.Close() }()
	for _, migration := range audit.Migrations() {
		for _, sql := range migration.Statements {
			if _, err := db.Exec(ctx, sql); err != nil {
				t.Fatal(err)
			}
		}
	}
	// Complete typed delivery probe on the REAL Tx, in addition to the REAL
	// DS summary recorder. This test table is not an accepted public migration.
	if _, err := db.Exec(ctx, `CREATE TABLE material_audit_probe(id INTEGER PRIMARY KEY,payload BLOB NOT NULL)`); err != nil {
		t.Fatal(err)
	}
	validator, err := audit.NewValidator([]string{"auth.setup", "auth.bootstrap", "auth.login", "auth.grant.create", "auth.grant.revoke", "auth.session.revoke", "xray.material.generate", "xray.material.reveal", "xray.material.use", "xray.material.revoke"}, audit.DefaultLimits())
	if err != nil {
		t.Fatal(err)
	}
	var recorder *audit.SQLRecorder
	var identity *auth.IdentityService
	var credentials *auth.CredentialService
	var materials *auth.XrayMaterialService
	var failMaterialAudit, omitMaterialAudit bool
	var beforeTransaction func() error
	approved := map[string]auth.XrayMaterialOperation{} // explicit task/confirmation/profile fixture, NOT a second scheduler
	var makeIdentity func(ds.Queryer, bool) *auth.IdentityService
	makeIdentity = func(q ds.Queryer, writable bool) *auth.IdentityService {
		backend := auth.IdentitySQLBackend{Read: statements(q).Read, InTx: func(context.Context, func(auth.IdentitySQLStatements) error) error {
			return auth.ErrIdentityUnavailable
		}}
		if writable {
			backend.InTxContext = func(ctx context.Context, fn func(context.Context, auth.IdentitySQLStatements) error) error {
				if beforeTransaction != nil {
					hook := beforeTransaction
					beforeTransaction = nil
					if err := hook(); err != nil {
						return err
					}
				}
				return db.InTx(ctx, func(ctx context.Context, tx ds.Tx) error {
					ops := statements(tx)
					ops.AppendAudit = func(ctx context.Context, events []auth.IdentityAuditEvent) error {
						for _, e := range events {
							if err := recorder.AppendTx(ctx, tx, audit.Event{SubjectID: e.SubjectID, Action: e.Action, Result: e.Result, RequestID: e.RequestID, Source: audit.Source(e.Source), NodeIDs: e.NodeIDs, At: e.At}); err != nil {
								return err
							}
						}
						return nil
					}
					if !omitMaterialAudit {
						ops.AppendXrayMaterialAudit = func(ctx context.Context, events []auth.XrayMaterialAuditEvent) error {
							for _, e := range events {
								raw, err := json.Marshal(e)
								if err != nil {
									return err
								}
								if _, err = tx.Exec(ctx, `INSERT INTO material_audit_probe(payload) VALUES(?)`, raw); err != nil {
									return err
								}
								result, reason := "success", ""
								if e.Result == "reserved" {
									result = "allowed"
								}
								if e.Result == "unknown" {
									result, reason = "failure", "xray_material_unknown"
								}
								if err = recorder.AppendTx(ctx, tx, audit.Event{SubjectID: e.SubjectID, Action: "xray.material." + e.Action, Result: result, RequestID: e.OperationID, Source: audit.Source(e.Source), NodeIDs: []string{e.Scope.NodeID}, ReasonCode: reason, At: e.At}); err != nil {
									return err
								}
							}
							if failMaterialAudit {
								return errors.New("injected after real typed and DS audit insert")
							}
							return nil
						}
					}
					return fn(context.WithValue(ctx, materialSQLReaderKey{}, makeIdentity(tx, false)), ops)
				})
			}
		}
		store, err := auth.NewSQLIdentityStore(backend)
		if err != nil {
			t.Fatal(err)
		}
		s, err := auth.NewIdentityServiceWithResources(store, func(_ context.Context, kind, id string) ([]string, error) {
			if kind == "node" && id == "node-a" {
				return []string{id}, nil
			}
			return nil, auth.ErrPolicyUnavailable
		}, auth.XrayMaterialPermissions())
		if err != nil {
			t.Fatal(err)
		}
		return s
	}
	bind := func() {
		var err error
		recorder, err = audit.NewSQLRecorder(db, validator, time.Now)
		if err != nil {
			t.Fatal(err)
		}
		identity = makeIdentity(db, true)
		credentials, err = auth.NewCredentialService(identity, "material-data-key", map[string][]byte{"material-data-key": bytes.Repeat([]byte{79}, 32)}, nil, nil)
		if err != nil {
			t.Fatal(err)
		}
		materials, err = auth.NewXrayMaterialService(credentials, func(ctx context.Context, check auth.XrayMaterialCheck) error {
			proof, ok := ctx.Value(materialSQLActorKey{}).(materialSQLActor)
			if !ok || proof.observation.SubjectID != check.Operation.SubjectID {
				return auth.ErrDenied
			}
			reader, inside := ctx.Value(materialSQLReaderKey{}).(*auth.IdentityService)
			if strings.HasPrefix(check.Stage, "commit-") && !inside {
				return auth.ErrIdentityUnavailable
			}
			if !inside {
				reader = identity
			}
			if err := reader.CheckObservationReadOnly(ctx, proof.secret, proof.observation); err != nil {
				return err
			}
			wanted, ok := approved[check.Operation.OperationID]
			if !ok || materialSQLDigest(wanted) != materialSQLDigest(check.Operation) || check.Operation.Scope.NodeID != "node-a" || (check.Operation.Scope.InstanceID != "instance-a" && check.Operation.Scope.InstanceID != "instance-b") {
				return auth.ErrDenied
			}
			return nil
		})
		if err != nil {
			t.Fatal(err)
		}
	}
	bind()
	setup, err := identity.PrepareSetup(ctx)
	if err != nil {
		t.Fatal(err)
	}
	if err = identity.Bootstrap(ctx, setup, "material-admin", "long-local-material-test-password"); err != nil {
		t.Fatal(err)
	}
	login := func() auth.LoginSession {
		s, err := identity.Login(ctx, "material-admin", "long-local-material-test-password", "127.0.0.1")
		if err != nil {
			t.Fatal(err)
		}
		return s
	}
	admin := login()
	actorContext := func(session auth.LoginSession) context.Context {
		observation, err := identity.Authenticate(ctx, session.Secret, session.CSRF, true)
		if err != nil {
			t.Fatal(err)
		}
		return context.WithValue(ctx, materialSQLActorKey{}, materialSQLActor{session.Secret, observation})
	}
	actorCtx := actorContext(admin)
	makeOp := func(action, id string, scope auth.XrayMaterialScope) auth.XrayMaterialOperation {
		ref, err := auth.XrayMaterialReference(scope)
		if err != nil {
			t.Fatal(err)
		}
		op := auth.XrayMaterialOperation{SubjectID: admin.Identity.SubjectID, OperationID: id, Action: action, Source: "web", Scope: scope, ReferenceID: ref, ExpiresAt: time.Now().Add(time.Minute)}
		if action != "generate" {
			op.ExpectedRevision = 1
		}
		if action == "reveal" {
			op.RecipientID = op.SubjectID
		}
		if action == "use" {
			op.Source = "plugin"
		}
		op.PlanDigest = materialSQLDigest(op)
		approved[id] = op
		return op
	}
	scope := auth.XrayMaterialScope{NodeID: "node-a", InstanceID: "instance-a", InboundID: "inbound-a", UserID: "user-a", Purpose: "xray-user"}
	generate := makeOp("generate", "generate-material-a", scope)
	if _, err = materials.Generate(actorCtx, generate); err != auth.ErrDenied {
		t.Fatal("real administrator got default node material grant", err)
	}
	permissions := []string{}
	for _, p := range auth.XrayMaterialPermissions() {
		permissions = append(permissions, p.Key)
	}
	if _, err = identity.CreateResourceGrant(ctx, admin.Secret, admin.CSRF, auth.GrantRequest{PrincipalID: admin.Identity.SubjectID, ResourceType: "node", ResourceID: "node-a", Permissions: permissions}); err != nil {
		t.Fatal(err)
	}
	actorCtx = actorContext(admin)
	omitMaterialAudit = true
	if _, err = materials.Generate(actorCtx, generate); err != auth.ErrIdentityUnavailable {
		t.Fatal("missing complete same-Tx sink did not close", err)
	}
	omitMaterialAudit = false
	meta, err := materials.Generate(actorCtx, generate)
	if err != nil || meta.State != "ready" {
		t.Fatal("actual central material generation failed", err)
	}
	// Export metadata for current user/config/plugin remains an explicit fixture;
	// material reference/revision, auth/grants, cipher and one-use are REAL SQL.
	request := xray.ClientExportRequest{RequestID: "actual-export-1", SubjectID: admin.Identity.SubjectID, RecipientID: admin.Identity.SubjectID, Source: "web", Format: "vless-uri", NodeID: scope.NodeID, InstanceID: scope.InstanceID, UserID: scope.UserID, ExpiresAt: time.Now().Add(time.Minute)}
	reader := func(ctx context.Context, req xray.ClientExportRequest) (xray.ClientExportSnapshot, error) {
		op := makeOp("metadata", "metadata-"+req.RequestID, scope)
		m, err := materials.Metadata(ctx, op)
		if err != nil {
			return xray.ClientExportSnapshot{}, err
		}
		return xray.ClientExportSnapshot{NodeID: scope.NodeID, InstanceID: scope.InstanceID, InboundID: scope.InboundID, UserID: scope.UserID, IdentityRef: m.ID, TemplateID: "vless-tls-vision", TargetVersion: xray.TestedXrayVersion, Host: "fleet.test", ServerName: "fleet.test", Label: "Local export", Port: 443, ConfigRevision: 1, UserRevision: 1, PluginRevision: 1, CredentialRevision: m.Revision, PluginEnabled: true, UserEnabled: m.State == "ready", ObservedAt: time.Now()}, nil
	}
	admit := func(ctx context.Context, b xray.ClientExportBinding) error {
		proof, ok := ctx.Value(materialSQLActorKey{}).(materialSQLActor)
		if !ok || b.Request.SubjectID != proof.observation.SubjectID || b.Snapshot.IdentityRef != meta.ID {
			return auth.ErrDenied
		}
		return identity.CheckObservationReadOnly(ctx, proof.secret, proof.observation)
	}
	var lastLoan auth.XrayMaterialOperation
	borrow := func(ctx context.Context, b xray.ClientExportBinding, use func([]byte) error) error {
		op := makeOp("reveal", b.Request.RequestID, scope)
		op.PlanDigest = b.DigestSHA256
		op.ExpiresAt = b.Request.ExpiresAt
		approved[op.OperationID] = op
		lastLoan = op
		return materials.WithMaterial(ctx, op, func(_ context.Context, raw []byte) error { return use(raw) })
	}
	deliveries := 0
	var clientUUID string
	if err = xray.WithClientExport(actorCtx, request, reader, admit, borrow, func(_ context.Context, _ string, raw []byte, _ time.Time) error {
		deliveries++
		u, err := url.Parse(string(raw))
		if err != nil {
			return err
		}
		clientUUID = u.User.Username()
		if len(clientUUID) != 36 || u.Hostname() != "fleet.test" || u.Query().Get("security") != "tls" {
			return errors.New("invalid generated connection")
		}
		return nil
	}); err != nil || deliveries != 1 {
		t.Fatal("actual credential-fenced export failed", err)
	}
	var payload []byte
	if err = db.QueryRow(ctx, `SELECT payload FROM cx_auth_identity WHERE singleton=1`).Scan(&payload); err != nil || bytes.Contains(payload, []byte(clientUUID)) {
		t.Fatal("UUID plaintext in auth SQL", err)
	}
	var auditRaw []byte
	if err = db.QueryRow(ctx, `SELECT payload FROM material_audit_probe ORDER BY id DESC LIMIT 1`).Scan(&auditRaw); err != nil {
		t.Fatal(err)
	}
	var event auth.XrayMaterialAuditEvent
	if json.Unmarshal(auditRaw, &event) != nil || event.Result != "delivered" || event.ReferenceID != meta.ID || event.Scope != scope || event.PlanDigest != lastLoan.PlanDigest || bytes.Contains(auditRaw, []byte(clientUUID)) {
		t.Fatal("complete material audit binding/secret protection failed")
	}
	if err = db.Close(); err != nil {
		t.Fatal(err)
	}
	db = newDB(t, dbPath, false)
	bind()
	if err = materials.WithMaterial(actorCtx, lastLoan, func(context.Context, []byte) error { t.Error("real SQL loan repeated after reopen"); return nil }); err != auth.ErrCredentialReplay {
		t.Fatal("SQL claim lost", err)
	}
	// Fill the export buffer, then fail the complete audit Tx: browser delivery
	// must remain zero, and the reserved SQL claim survives another restart.
	request.RequestID = "actual-export-audit-fail"
	failedBorrow := func(ctx context.Context, b xray.ClientExportBinding, use func([]byte) error) error {
		op := makeOp("reveal", b.Request.RequestID, scope)
		op.PlanDigest = b.DigestSHA256
		op.ExpiresAt = b.Request.ExpiresAt
		approved[op.OperationID] = op
		lastLoan = op
		return materials.WithMaterial(ctx, op, func(_ context.Context, raw []byte) error {
			if err := use(raw); err != nil {
				return err
			}
			failMaterialAudit = true
			return nil
		})
	}
	if err = xray.WithClientExport(actorCtx, request, reader, admit, failedBorrow, func(context.Context, string, []byte, time.Time) error { deliveries++; return nil }); err != xray.ErrClientExportUnknown || deliveries != 1 {
		t.Fatal("secret delivered before real audit commit", err)
	}
	failMaterialAudit = false
	if err = db.Close(); err != nil {
		t.Fatal(err)
	}
	db = newDB(t, dbPath, false)
	bind()
	if err = materials.WithMaterial(actorCtx, lastLoan, func(context.Context, []byte) error { t.Error("unknown SQL export retried"); return nil }); err != auth.ErrCredentialReplay {
		t.Fatal("unknown claim became retryable", err)
	}
	if generic, err := credentials.List(ctx, admin.Secret); err != nil || len(generic) != 0 {
		t.Fatal("typed Xray material entered generic public credential kinds", err)
	}
	// Actual pinned R36 auth production source must refuse this newer payload;
	// a prior writer must not silently drop private materials/permanent claims.
	priorOps := func(q ds.Queryer) legacy.IdentitySQLStatements {
		ops := statements(q)
		return legacy.IdentitySQLStatements{Read: ops.Read, Exec: ops.Exec}
	}
	priorStore, err := legacy.NewSQLIdentityStore(legacy.IdentitySQLBackend{Read: priorOps(db).Read, InTx: func(ctx context.Context, fn func(legacy.IdentitySQLStatements) error) error {
		return db.InTx(ctx, func(ctx context.Context, tx ds.Tx) error { return fn(priorOps(tx)) })
	}})
	if err != nil {
		t.Fatal(err)
	}
	prior, err := legacy.NewIdentityService(priorStore)
	if err != nil {
		t.Fatal(err)
	}
	var beforeOld, afterOld []byte
	if err = db.QueryRow(ctx, `SELECT payload FROM cx_auth_identity WHERE singleton=1`).Scan(&beforeOld); err != nil {
		t.Fatal(err)
	}
	if _, err = prior.PrepareSetup(ctx); err != legacy.ErrIdentityUnavailable {
		t.Fatal("old real writer accepted material payload instead of failing closed", err)
	}
	if err = db.QueryRow(ctx, `SELECT payload FROM cx_auth_identity WHERE singleton=1`).Scan(&afterOld); err != nil || !bytes.Equal(beforeOld, afterOld) {
		t.Fatal("old writer erased material/fences", err)
	}
	// Real authenticated session revocation between entry and BEGIN is observed
	// by the SAME Tx reader and cannot allocate a second user identity.
	otherScope := scope
	otherScope.UserID = "user-b"
	other := makeOp("generate", "generate-after-revoke", otherScope)
	beforeTransaction = func() error { return identity.RevokeSession(ctx, admin.Secret, admin.CSRF, admin.Identity.SessionID) }
	if _, err = materials.Generate(actorCtx, other); err != auth.ErrDenied {
		t.Fatal("revoked session generated material after entry", err)
	}
	if _, err = materials.Generate(ctx, other); err != auth.ErrDenied {
		t.Fatal("known subject/ref acted as source credential", err)
	}
}
