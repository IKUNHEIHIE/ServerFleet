type materialSQLReaderKey struct{}
type materialSQLActorKey struct{}
type materialSQLActor struct {
	secret      string
	observation auth.Identity
}

// Isolated Host storage/audit adapter on the actual DS transaction. These probe
// tables do not allocate a public migration or grant a plugin raw SQL access.
type managedUsersSQLStore struct {
	db          *ds.DB
	scope       xray.ManagedUserScope
	bindReader  func(ds.Queryer) *auth.IdentityService
	failAudit   *bool
	recorder    *audit.SQLRecorder
	beforeBegin func() error
}

func (s *managedUsersSQLStore) View(ctx context.Context, fn func(context.Context, []byte) error) error {
	return s.db.InTx(ctx, func(ctx context.Context, tx ds.Tx) error {
		var raw []byte
		if err := tx.QueryRow(ctx, `SELECT payload FROM managed_user_probe WHERE scope=?`, materialSQLDigest(s.scope)).Scan(&raw); err != nil {
			return err
		}
		return fn(context.WithValue(ctx, materialSQLReaderKey{}, s.bindReader(tx)), raw)
	})
}
func (s *managedUsersSQLStore) Update(ctx context.Context, fn func(context.Context, []byte) ([]byte, []xray.ManagedUserAudit, error)) error {
	if s.beforeBegin != nil {
		hook := s.beforeBegin
		s.beforeBegin = nil
		if err := hook(); err != nil {
			return err
		}
	}
	return s.db.InTx(ctx, func(ctx context.Context, tx ds.Tx) error {
		var raw []byte
		if err := tx.QueryRow(ctx, `SELECT payload FROM managed_user_probe WHERE scope=?`, materialSQLDigest(s.scope)).Scan(&raw); err != nil {
			return err
		}
		next, events, err := fn(context.WithValue(ctx, materialSQLReaderKey{}, s.bindReader(tx)), raw)
		if err != nil {
			return err
		}
		for _, e := range events {
			payload, err := json.Marshal(e)
			if err != nil {
				return err
			}
			if _, err = tx.Exec(ctx, `INSERT INTO managed_user_audit_probe(payload) VALUES(?)`, payload); err != nil {
				return err
			}
			result, reason := "success", ""
			if e.Result == "pending" {
				result = "allowed"
			}
			if e.Result == "unknown" {
				result, reason = "failure", "xray_user_unknown"
			}
			if err = s.recorder.AppendTx(ctx, tx, audit.Event{SubjectID: e.Operation.SubjectID, Action: "xray.user." + e.Operation.Action, Result: result, ReasonCode: reason, RequestID: e.Operation.OperationID, Source: audit.Source(e.Operation.Source), NodeIDs: []string{e.Operation.Scope.NodeID}, At: e.At}); err != nil {
				return err
			}
		}
		if *s.failAudit {
			return errors.New("injected after actual user audit insert")
		}
		result, err := tx.Exec(ctx, `UPDATE managed_user_probe SET payload=? WHERE scope=?`, next, materialSQLDigest(s.scope))
		if err != nil {
			return err
		}
		count, err := result.RowsAffected()
		if err != nil {
			return err
		}
		if count != 1 {
			return xray.ErrManagedUser
		}
		return nil
	})
}

func materialSQLDigest(v any) string {
	raw, _ := json.Marshal(v)
	hash := sha256.Sum256(raw)
	return hex.EncodeToString(hash[:])
}

	}
	// Actual pinned R36 auth production source must refuse this newer payload;
	// a prior writer must not silently drop private materials/permanent claims.
	priorOps := func(q ds.Queryer) legacy.IdentitySQLStatements {
		ops := statements(q)
		return legacy.IdentitySQLStatements{Read: ops.Read, Exec: ops.Exec}
	}
	priorStore, err := legacy.NewSQLIdentityStore(legacy.IdentitySQLBackend{Read: priorOps(db).Read, InTx: func(ctx context.Context, fn func(legacy.IdentitySQLStatements) error) error {
		return db.InTx(ctx, func(ctx context.Context, tx ds.Tx) error { return fn(priorOps(tx)) })
	}})
	if err != nil {
		t.Fatal(err)
	}
	prior, err := legacy.NewIdentityService(priorStore)
	if err != nil {
		t.Fatal(err)
	}
	var beforeOld, afterOld []byte
	if err = db.QueryRow(ctx, `SELECT payload FROM cx_auth_identity WHERE singleton=1`).Scan(&beforeOld); err != nil {
		t.Fatal(err)
	}
	if _, err = prior.PrepareSetup(ctx); err != legacy.ErrIdentityUnavailable {
		t.Fatal("old real writer accepted material payload instead of failing closed", err)
	}
	if err = db.QueryRow(ctx, `SELECT payload FROM cx_auth_identity WHERE singleton=1`).Scan(&afterOld); err != nil || !bytes.Equal(beforeOld, afterOld) {
		t.Fatal("old writer erased material/fences", err)
	}
	// Real persistent rename preserves attribution; a failed same-Tx user audit
	// changes neither the metadata nor either audit stream.
	rename := userOperation("rename", "rename-user-a", managed.Revision)
	renameChange := xray.ManagedUserChange{DisplayName: "Renamed user"}
	approveUser(rename, renameChange, xray.ManagedUserApplication{})
	var beforeUser, afterUser []byte
	var beforeAudits, afterAudits int
	if err = db.QueryRow(ctx, `SELECT payload FROM managed_user_probe WHERE scope=?`, materialSQLDigest(userScope)).Scan(&beforeUser); err != nil {
		t.Fatal(err)
	}
	if bytes.Contains(beforeUser, []byte(clientUUID)) {
		t.Fatal("persistent user metadata contains plaintext UUID")
	}
	if err = db.QueryRow(ctx, `SELECT count(*) FROM managed_user_audit_probe`).Scan(&beforeAudits); err != nil {
		t.Fatal(err)
	}
	failUserAudit = true
	if got, err := users.Change(actorCtx, rename, renameChange); err == nil || got.ID != "" {
		t.Fatal("failed user audit reported success", err)
	}
	failUserAudit = false
	if err = db.QueryRow(ctx, `SELECT payload FROM managed_user_probe WHERE scope=?`, materialSQLDigest(userScope)).Scan(&afterUser); err != nil || !bytes.Equal(beforeUser, afterUser) {
		t.Fatal("user write survived rolled-back audit", err)
	}
	if err = db.QueryRow(ctx, `SELECT count(*) FROM managed_user_audit_probe`).Scan(&afterAudits); err != nil || afterAudits != beforeAudits {
		t.Fatal("user audit survived failed transaction", err)
	}
	renamed, err := users.Change(actorCtx, rename, renameChange)
	if err != nil || renamed.StatsEmail != managed.StatsEmail || renamed.IdentityRef != managed.IdentityRef || renamed.Application != "pending" {
		t.Fatal("real rename lost identity/attribution", err)
	}
	managed = renamed
	confirmRename := userOperation("application", "confirm-rename-user-a", managed.Revision)
	confirmRename.TaskID, confirmRename.PlanDigest = managed.PendingTaskID, managed.PendingPlanDigest
	approveUser(confirmRename, xray.ManagedUserChange{}, userEvidence)
	managed, err = users.RecordApplication(actorCtx, confirmRename, userEvidence)
	if err != nil {
		t.Fatal(err)
	}
	request.RequestID = "renamed-user-export"
	if err = xray.WithClientExport(actorCtx, request, reader, admit, borrow, func(_ context.Context, _ string, raw []byte, _ time.Time) error {
		u, err := url.Parse(string(raw))
		if err != nil {
			return err
		}
		if u.User.Username() != clientUUID || u.Fragment != "Renamed user" {
			return errors.New("rename changed identity or lost label")
		}
		deliveries++
		return nil
	}); err != nil || deliveries != 2 {
		t.Fatal("real renamed user export failed", err)
	}
	disable := userOperation("set-enabled", "disable-user-a", managed.Revision)
	approveUser(disable, xray.ManagedUserChange{}, xray.ManagedUserApplication{})
	managed, err = users.Change(actorCtx, disable, xray.ManagedUserChange{})
	if err != nil || managed.Enabled || managed.Application != "pending" {
		t.Fatal("real disable was falsely acknowledged applied", err)
	}
	if err = db.Close(); err != nil {
		t.Fatal(err)
	}
	db = newDB(t, dbPath, false)
	bind()
	request.RequestID = "disabled-user-export"
	if err = xray.WithClientExport(actorCtx, request, reader, admit, borrow, func(context.Context, string, []byte, time.Time) error { t.Error("disabled user delivered"); return nil }); err != xray.ErrClientExport {
		t.Fatal("disabled/pending state was revived by reopen", err)
	}
	listOp := userOperation("list", "list-persistent-users", 0)
	listOp.TaskID, listOp.UserID = "", ""
	pageRequest := xray.ManagedUserPageRequest{Limit: 1}
	approvedUsers[listOp.OperationID] = xray.ManagedUserCheck{Operation: listOp, Page: pageRequest}
	page, err := users.List(actorCtx, listOp, pageRequest)
	if err != nil || len(page.Users) != 1 || page.Users[0].Enabled || page.Users[0].Application != "pending" || page.Users[0].DisplayName != "Renamed user" || page.Users[0].StatsEmail != managed.StatsEmail {
		t.Fatal("actual SQL metadata page lost current user state", err)
	}
	// A REAL session revoked after entry but before user BEGIN must be rejected
	// by the transaction-bound reader; the pending disable remains unchanged.
	staleRename := userOperation("rename", "rename-after-revoke", managed.Revision)
	approveUser(staleRename, xray.ManagedUserChange{DisplayName: "Revoked"}, xray.ManagedUserApplication{})
	userStore.beforeBegin = func() error { return identity.RevokeSession(ctx, admin.Secret, admin.CSRF, admin.Identity.SessionID) }
	if _, err = users.Change(actorCtx, staleRename, xray.ManagedUserChange{DisplayName: "Revoked"}); err == nil {
		t.Fatal("revoked real session mutated user after entry")
	}
	admin = login()
	actorCtx = actorContext(admin)
	// Real authenticated session revocation between entry and BEGIN is observed
	// by the SAME Tx reader and cannot allocate a second user identity.
	otherScope := scope
	otherScope.UserID = "user-b"
	other := makeOp("generate", "generate-after-revoke", otherScope)
	beforeTransaction = func() error { return identity.RevokeSession(ctx, admin.Secret, admin.CSRF, admin.Identity.SessionID) }
	if _, err = materials.Generate(actorCtx, other); err != auth.ErrDenied {
		t.Fatal("revoked session generated material after entry", err)
	}
	if _, err = materials.Generate(ctx, other); err != auth.ErrDenied {
		t.Fatal("known subject/ref acted as source credential", err)
	}
}
